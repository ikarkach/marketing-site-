{{- /* Хлебные крошки + разметка BreadcrumbList. Вход: страница. Категория берётся из front matter (category, categoryUrl). */ -}} {{- $p := . -}} {{- $crumbs := slice (dict "t" "Главная" "u" "/") -}} {{- if eq $p.Section "projects" -}} {{- if ne $p.RelPermalink "/projects/" }}{{ $crumbs = $crumbs | append (dict "t" "Каталог" "u" "/projects/") }}{{ end -}} {{- with $p.Params.category }}{{ $crumbs = $crumbs | append (dict "t" . "u" $p.Params.categoryUrl) }}{{ end -}} {{- /* страница модели: перед названием модели — страница серии/типа, к которой она относится */ -}} {{- with $p.Params.parentPage }}{{ with site.GetPage . }}{{ $crumbs = $crumbs | append (dict "t" (.Params.crumb | default (.Title | plainify)) "u" .RelPermalink) }}{{ end }}{{ end -}} {{- else if and $p.IsPage (eq $p.Section "blog") -}} {{- $crumbs = $crumbs | append (dict "t" "Блог" "u" "/blog/") -}} {{- end -}} {{- $crumbs = $crumbs | append (dict "t" ($p.Params.crumb | default ($p.Title | plainify)) "u" $p.RelPermalink) -}}
    {{- range $i, $c := $crumbs }} {{- if eq (add $i 1) (len $crumbs) }}
  1. {{ $c.t }}
    2. {{- else }}
  2. {{ $c.t }}
    3. {{- end }} {{- end }}
{{- $items := slice }} {{- range $i, $c := $crumbs }}{{ $items = $items | append (dict "@type" "ListItem" "position" (add $i 1) "name" $c.t "item" ($c.u | absURL)) }}{{ end }}
